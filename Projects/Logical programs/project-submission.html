function isPalindrome(str) {
  const clean = String(str).toLowerCase().replace(/[^a-z0-9]/g, '');
  return clean === clean.split('').reverse().join('');
}
02

Fibonacci Sequence Generator

Time: O(N) | Space: O(N)
Sequence

Generates $N$ numbers in the Fibonacci sequence where $F(n) = F(n-1) + F(n-2)$, starting with 0 and 1.

Click "Generate Series" to view numbers... 
function generateFibonacci(n) {
  if (n <= 0) return [];
  if (n === 1) return [0];
  const seq = [0, 1];
  for (let i = 2; i < n; i++) {
    seq.push(seq[i - 1] + seq[i - 2]);
  }
  return seq;
}

function isPrime(num) {
  if (num <= 1) return false;
  for (let i = 2; i <= Math.sqrt(num); i++) {
    if (num % i === 0) return false;
  }
  return true;
}

05

Factorial & Recursion

Time: O(N) | BigInt Support
Recursion

Computes $N! = N \times (N-1) \dots \times 1$ with step breakdown and handles arbitrary precision values.

Click "Calculate Factorial" to compute... 
function factorial(n) {
  if (n < 0) return null;
  if (n === 0 || n === 1) return 1n;
  let result = 1n;
  for (let i = 2n; i <= BigInt(n); i++) {
    result *= i;
  }
  return result;
}
06

FizzBuzz Challenge

Time: O(N) | Modulo Logic
Classic

Prints multiples of 3 as "Fizz", 5 as "Buzz", and both as "FizzBuzz" with color-coded token badges.

Click "Run FizzBuzz" to generate... 
function fizzBuzz(limit) {
  const res = [];
  for (let i = 1; i <= limit; i++) {
    if (i % 15 === 0) res.push('FizzBuzz');
    else if (i % 3 === 0) res.push('Fizz');
    else if (i % 5 === 0) res.push('Buzz');
    else res.push(i);
  }
  return res;
}

function bubbleSort(arr) {
  const a = [...arr];
  for (let i = 0; i < a.length; i++) {
    for (let j = 0; j < a.length - i - 1; j++) {
      if (a[j] > a[j + 1]) [a[j], a[j + 1]] = [a[j + 1], a[j]];
    }
  }
  return a;
}
09

Text Statistics Analyzer

Vowels, Consonants, Words
NLP / Regex

Parses raw text streams and outputs exact counts of vowels, consonants, digits, words, and estimated reading time.

Click "Analyze Text" to calculate breakdown... 
function analyzeText(text) {
  const vowels = (text.match(/[aeiou]/gi) || []).length;
  const consonants = (text.match(/[bcdfghjklmnpqrstvwxyz]/gi) || []).length;
  const words = text.trim() ? text.trim().split(/\s+/).length : 0;
  return { vowels, consonants, words };
}

Pattern & Matrix Generator

Nested Loops | Matrix Art
Patterns

Renders complex 2D geometric patterns (Pyramid, Diamond, Floyd's Triangle, Number Matrix) using algorithmic looping. 

Click "Render Pattern" to output...
function generatePyramid(rows) {
  let out = '';
  for (let i = 1; i <= rows; i++) {
    out += ' '.repeat((rows - i) * 2) + '* '.repeat(i * 2 - 1) + '\n';
  }
  return out;
}
